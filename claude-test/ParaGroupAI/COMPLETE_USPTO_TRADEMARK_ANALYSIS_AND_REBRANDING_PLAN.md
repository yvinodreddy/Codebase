Create production-ready script to rename all files:

```bash
#!/bin/bash
# rename_claudeprompt_to_paragroup.sh
# PRODUCTION-READY automated renaming script
# Created: 2025-11-28

set -e  # Exit on error
set -u  # Exit on undefined variable

echo "=================================="
echo "Para Group Rebranding Script v1.0"
echo "=================================="
echo "This script will rename all ClaudePrompt references to Para Group AI Orchestrator"
echo ""
read -p "Continue? (yes/no): " confirm
if [ "$confirm" != "yes" ]; then
    echo "Aborted."
    exit 1
fi

# Create backup
BACKUP_DIR="backup_before_rebrand_$(date +%Y%m%d_%H%M%S)"
echo "Creating backup in $BACKUP_DIR..."
mkdir -p "$BACKUP_DIR"
tar -czf "$BACKUP_DIR/full_backup.tar.gz" . --exclude="$BACKUP_DIR"

# Log file
LOG_FILE="rebrand_log_$(date +%Y%m%d_%H%M%S).txt"
echo "Logging to $LOG_FILE"

# Function to rename files
rename_files() {
    echo "Step 1: Renaming files containing 'ClaudePrompt'..."
    find . -type f -name "*ClaudePrompt*" -not -path "./$BACKUP_DIR/*" | while read file; do
        newname=$(echo "$file" | sed 's/ClaudePrompt/ParaGroupAIOrchestrator/g')
        if [ "$file" != "$newname" ]; then
            echo "Renaming: $file → $newname" | tee -a "$LOG_FILE"
            mkdir -p "$(dirname "$newname")"
            mv "$file" "$newname"
        fi
    done
    echo "✓ File renaming complete"
}

# Function to update file contents
update_contents() {
    echo "Step 2: Updating file contents..."

    # Python files
    echo "  Updating Python files..."
    find . -type f -name "*.py" -not -path "./$BACKUP_DIR/*" -exec sed -i \
        -e 's/ClaudePrompt/ParaGroupAIOrchestrator/g' \
        -e 's/claudeprompt/paragroupaiorchestrator/g' \
        -e 's/CLAUDEPROMPT/PARAGROUPAIORCHESTRATOR/g' \
        {} \; 2>&1 | tee -a "$LOG_FILE"

    # Markdown files
    echo "  Updating Markdown files..."
    find . -type f -name "*.md" -not -path "./$BACKUP_DIR/*" -exec sed -i \
        -e 's/ClaudePrompt/Para Group AI Orchestrator®/g' \
        -e 's/claudeprompt/paragroup-ai-orchestrator/g' \
        {} \; 2>&1 | tee -a "$LOG_FILE"

    # Config files (JSON, YAML, TOML)
    echo "  Updating config files..."
    find . -type f \( -name "*.json" -o -name "*.yaml" -o -name "*.yml" -o -name "*.toml" \) \
        -not -path "./$BACKUP_DIR/*" -exec sed -i \
        -e 's/ClaudePrompt/ParaGroupAIOrchestrator/g' \
        -e 's/claudeprompt/paragroup-ai-orchestrator/g' \
        {} \; 2>&1 | tee -a "$LOG_FILE"

    # Shell scripts
    echo "  Updating shell scripts..."
    find . -type f -name "*.sh" -not -path "./$BACKUP_DIR/*" -exec sed -i \
        -e 's/ClaudePrompt/ParaGroupAIOrchestrator/g' \
        -e 's/claudeprompt/paragroupaiorchestrator/g' \
        {} \; 2>&1 | tee -a "$LOG_FILE"

    # README and documentation
    echo "  Updating documentation..."
    find . -type f \( -name "README*" -o -name "*.txt" \) -not -path "./$BACKUP_DIR/*" -exec sed -i \
        -e 's/ClaudePrompt/Para Group AI Orchestrator®/g' \
        {} \; 2>&1 | tee -a "$LOG_FILE"

    echo "✓ Content update complete"
}

# Function to update package names
update_packages() {
    echo "Step 3: Updating package configurations..."

    # setup.py
    if [ -f "setup.py" ]; then
        echo "  Updating setup.py..." | tee -a "$LOG_FILE"
        sed -i \
            -e 's/name="claudeprompt"/name="paragroup-ai-orchestrator"/g' \
            -e 's/ClaudePrompt/Para Group AI Orchestrator/g' \
            setup.py
    fi

    # pyproject.toml
    if [ -f "pyproject.toml" ]; then
        echo "  Updating pyproject.toml..." | tee -a "$LOG_FILE"
        sed -i \
            -e 's/name = "claudeprompt"/name = "paragroup-ai-orchestrator"/g' \
            -e 's/ClaudePrompt/Para Group AI Orchestrator/g' \
            pyproject.toml
    fi

    # package.json (if exists)
    if [ -f "package.json" ]; then
        echo "  Updating package.json..." | tee -a "$LOG_FILE"
        sed -i \
            -e 's/"name": "claudeprompt"/"name": "paragroup-ai-orchestrator"/g' \
            -e 's/ClaudePrompt/Para Group AI Orchestrator/g' \

    echo "✓ Package update complete"
}

# Function to update imports
update_imports() {
```
