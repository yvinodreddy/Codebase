# What this does: Adds "margin of error" to your predictions

def predict_with_uncertainty(prompt, n_samples=100):
    """
    EXAMPLE: User asks "How long will this task take?"
    
    Instead of: "5 days" (99.3% confidence)
    You return: "5 days ± 1 day" (95% confidence: 4-6 days)
    
    This is done by running the prediction 100 times with slight variations
    and measuring the spread of answers.
    """
    predictions = []
    for _ in range(n_samples):
        pred = model.predict(prompt, dropout=True)
        predictions.append(pred)
    
    mean = np.mean(predictions)
    std = np.std(predictions)
    
    return {
        "prediction": mean,
        "confidence_interval_95": (mean - 1.96*std, mean + 1.96*std),
        "uncertainty": std
    }

# REAL-WORLD EXAMPLE:
# User: "Estimate project timeline"
# Without uncertainty: "30 days" (99.3% confidence)
# With uncertainty: "30 days ± 3 days (95% CI: 27-33 days)"
# Now user knows the range!
```

# What this does: Auto-redacts personally identifiable information

class PIIRedactionGuardrail:
    """
    EXAMPLE: User says "Contact John Smith at john@email.com, SSN 123-45-6789"
    
    This guardrail auto-redacts:
    "Contact John Smith at [REDACTED_EMAIL], SSN [REDACTED_SSN]"
    """
    
    PII_PATTERNS = {
        "ssn": r'\b\d{3}-\d{2}-\d{4}\b',
        "email": r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',
        "phone": r'\b\d{3}[-.]?\d{3}[-.]?\d{4}\b',
        "credit_card": r'\b\d{4}[-\s]?\d{4}[-\s]?\d{4}[-\s]?\d{4}\b'
    }
    
    def redact_pii(self, text):
        for pii_type, pattern in self.PII_PATTERNS.items():
            text = re.sub(pattern, f"[REDACTED_{pii_type.upper()}]", text)
        return text

# REAL-WORLD EXAMPLE:
# Input: "My SSN is 123-45-6789 and email is john@example.com"
# Output: "My SSN is [REDACTED_SSN] and email is [REDACTED_EMAIL]"

# What this does: Smarter context retrieval using meaning, not just keywords

from sentence_transformers import SentenceTransformer

class SemanticContextRetriever:
    """
    EXAMPLE: User asks "How did we handle user authentication?"
    
    Keyword search would look for: "user", "authentication"
    Semantic search understands:
    - "login" = authentication
    - "sign in" = authentication
    - "credentials" = authentication
    
    Finds MORE relevant results even without exact keywords!
    """
    
    def __init__(self):
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
    
    def retrieve_relevant_context(self, query, k=10):
        """Retrieve top-k most relevant messages by MEANING."""
        
        # Convert query to vector (semantic representation)
        query_embedding = self.model.encode(query)
        
        # Convert all messages to vectors
        message_embeddings = [
            self.model.encode(msg.content) for msg in self.messages
        ]
        
        # Calculate similarity (how similar is the meaning?)
        similarities = cosine_similarity([query_embedding], message_embeddings)[0]
        
        # Get top-k most similar
        top_k_indices = np.argsort(similarities)[-k:][::-1]
        
        return [self.messages[i] for i in top_k_indices]

# REAL-WORLD EXAMPLE:
# Keyword search:
#   Query: "login security"
#   Finds: Messages containing "login" AND "security"
#   Misses: Messages about "authentication" or "credentials"

# Semantic search:
#   Query: "login security"
#   Finds: Messages about login, authentication, credentials, sign-in, user verification
#   Much smarter!

# What this does: Compresses context while preserving meaning

class ContextCompressor:
    """
    EXAMPLE: Long message (1000 tokens) can be compressed to 500 tokens
    while keeping the important information.
    
    Original (1000 tokens):
    "The authentication system we implemented uses JWT tokens for session 
    management. We chose JWT because it's stateless, scalable, and widely 
    supported. The implementation includes refresh tokens for security, 
    token expiration after 15 minutes, and secure HTTP-only cookies to 
    prevent XSS attacks. We also added rate limiting to prevent brute 
    force attacks..."
    
    Compressed (500 tokens):
    "Authentication uses JWT tokens (stateless, scalable). Includes refresh 
    tokens, 15-min expiration, HTTP-only cookies (XSS prevention), rate 
    limiting (brute force prevention)..."
    
    Same information, 50% less space!
    """
    
    def compress(self, text, compression_ratio=0.5):
        """Compress text to target ratio while preserving meaning."""
        
        # 1. Split into tokens
        tokens = self.tokenize(text)
        
        # 2. Calculate importance of each token
        importance_scores = self.calculate_importance(tokens)
        
        # 3. Keep top tokens by importance
        target_tokens = int(len(tokens) * compression_ratio)
        top_tokens = self.select_top_tokens(tokens, importance_scores, target_tokens)
        
        # 4. Reconstruct compressed text
        compressed = self.reconstruct(top_tokens)
        
        return {
            "original_length": len(tokens),
            "compressed_length": len(top_tokens),
            "compression_ratio": len(top_tokens) / len(tokens),
            "compressed_text": compressed
        }

# REAL-WORLD EXAMPLE:
# Original conversation: 200,000 tokens (at compaction threshold)
# With compression: 100,000 tokens (50% compression)
# Benefit: Can store 2x more in memory before compaction
# Quality: 90%+ meaning preserved
```
